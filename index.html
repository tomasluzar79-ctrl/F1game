/* ROZSIRENI HRY: zivoty, checkpointy, vrtulniky */ let lives = 3; let checkpoints = []; let activeCheckpoint = 70; let helicopters = []; let helicopterShots = []; let helicopterClock = 0; let gameOver = false; const originalMission = mission; const originalUpdate = update; const originalDraw = draw; const originalHud = hud; const originalPanel = panel; // Zacatek mise mission = function(n) { originalMission(n); lives = 3; gameOver = false; activeCheckpoint = 70; helicopterClock = 0; helicopterShots = []; // Vlajky v kazde misi checkpoints = []; for (let x = 850; x < end - 250; x += 950) { checkpoints.push({ x, activated: false }); } // Vrtulniky - ve vyssich misich vice helicopters = []; for (let x = 650; x < end - 150; x += 650 - n * 65) { helicopters.push({ x, home: x, y: floor - 190 - Math.random() * 55, hp: 65 + n * 20, dir: -1, cooldown: 1.5 + Math.random() * 2, phase: Math.random() * 6.28, attacking: helicopters.length % 2 === 0 }); } // Posileni pozemnich nepratel foes.forEach((enemy, i) => { enemy.hp *= 1.25 + n * 0.15; enemy.max = enemy.hp; enemy.cd *= 0.8; // Dodatecne prekazky - hlidky if (i % 4 === 0) { enemy.role = 4; } }); updateLives(); }; // Zobrazeni zivotu function updateLives() { let el = document.getElementById("livesDisplay"); if (!el) { el = document.createElement("span"); el.id = "livesDisplay"; el.className = "plate"; document.getElementById("hud") .insertBefore(el, document.getElementById("ammo")); } el.textContent = "❤️ " + lives; } hud = function() { originalHud(); updateLives(); }; // Ztrata zivota a navrat na checkpoint function loseLife() { if (gameOver || mode !== "run") return; lives--; updateLives(); shake = 8; if (lives <= 0) { gameOver = true; originalPanel("lost"); $("title").textContent = "GAME OVER"; $("summary").textContent = "Ztratil jsi vsechny 3 zivoty. Zkus misi znovu."; $("action").textContent = "Hrát znovu"; return; } ride = null; p.x = activeCheckpoint; p.y = floor; p.vx = 0; p.vy = 0; p.hp = 100; p.safe = 2.5; p.ground = true; p.grace = 0.1; shots = []; helicopterShots = []; message("Ztratil jsi zivot! Zbyva: " + lives, 2); originalHud(); } // Zachyceni smrti z puvodni hry panel = function(state) { if (state === "lost" && !gameOver) { loseLife(); return; } originalPanel(state); }; // Aktualizace hernich mechanik update = function(dt) { if (mode !== "run") return; originalUpdate(dt); if (mode !== "run") return; // Checkpointy for (const cp of checkpoints) { if (!cp.activated && p.x >= cp.x) { cp.activated = true; activeCheckpoint = cp.x; message("🚩 CHECKPOINT ULOZEN", 2); } } helicopterClock += dt; for (const h of helicopters) { if (h.hp <= 0) continue; // Let vrtulniku h.x = h.home + Math.sin(helicopterClock * 0.65 + h.phase) * 95; h.y += Math.sin(helicopterClock * 2 + h.phase) * 12 * dt; h.dir = p.x < h.x ? -1 : 1; h.cooldown -= dt; // Utok vrtulniku if (h.attacking && Math.abs(p.x - h.x) < 450 && h.cooldown <= 0) { helicopterShots.push({ x: h.x, y: h.y + 20, vx: (p.x - h.x) * 0.65, vy: 155, life: 3 }); h.cooldown = 1.8 + Math.random() * 1.2; } } // Zasahy vrtulniku hracovymi strelami for (const s of shots) { if (s.enemy || s.life <= 0) continue; for (const h of helicopters) { if (h.hp <= 0) continue; if (Math.abs(s.x - h.x) < 43 && Math.abs(s.y - h.y) < 25) { h.hp -= s.w >= 4 ? 80 : 25; s.life = 0; puff(s.x, s.y, "#ffb347", 9, 1.2); if (h.hp <= 0) { puff(h.x, h.y, "#ff9030", 35, 2); kills++; shake = 7; } break; } } } // Strely vrtulniku for (const s of helicopterShots) { s.x += s.vx * dt; s.y += s.vy * dt; s.life -= dt; if (Math.abs(s.x - p.x) < 17 && Math.abs(s.y - (p.y - 28)) < 32) { hurt(22); s.life = 0; puff(s.x, s.y, "#ff9a40", 12, 1.2); } if (s.y >= floor) { s.life = 0; puff(s.x, floor, "#b98a55", 8, 1); } } helicopterShots = helicopterShots.filter(s => s.life > 0); }; // Vykresleni novych objektu draw = function() { originalDraw(); if (mode === "menu") return; g.save(); g.translate(0, height * 0.82 - floor * zoom); g.scale(zoom, zoom); g.translate(-cam, 0); // Vlajky checkpointu for (const cp of checkpoints) { rect(cp.x, floor - 90, 4, 90, "#b5b5a0"); rect( cp.x + 4, floor - 90, 35, 22, cp.activated ? "#49d46a" : "#e5b54d" ); g.fillStyle = "#ffffff"; g.font = "bold 12px monospace"; g.fillText("CP", cp.x + 8, floor - 74); } // Vrtulniky for (const h of helicopters) { if (h.hp <= 0) continue; g.save(); g.translate(h.x, h.y); g.scale(h.dir, 1); // Ocas rect(-62, -7, 48, 8, "#596c61"); rect(-65, -20, 5, 29, "#87978c"); // Trup ellipse(0, 0, 39, 17, "#536b60"); ellipse(15, -2, 17, 11, "#8ba7a6"); // Podvozek rect(-23, 17, 48, 3, "#303b38"); rect(-18, 10, 3, 9, "#303b38"); rect(18, 10, 3, 9, "#303b38"); // Rotor rect(-2, -23, 4, 9, "#363d39"); const rotor = Math.sin(clock * 40) * 9; rect(-53 + rotor, -25, 106 - rotor * 2, 3, "#c6d0c4"); // Utočny vrtulnik if (h.attacking) { rect(-12, 13, 23, 7, "#343a34"); rect(0, 20, 5, 10, "#292e2c"); } g.restore(); // Ukazatel zdravi vrtulniku rect(h.x - 25, h.y - 39, 50, 4, "#602d29"); rect(h.x - 25, h.y - 39, 50 * clamp(h.hp / (65 + level * 20), 0, 1), 4, "#8fcb78"); } // Projektily vrtulniku for (const s of helicopterShots) { ellipse(s.x, s.y, 5, 7, "#ffb34d"); ellipse(s.x, s.y, 2, 4, "#fff1b4"); } g.restore(); };
